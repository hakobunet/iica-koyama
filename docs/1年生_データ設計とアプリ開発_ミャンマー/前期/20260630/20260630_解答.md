## ပုစ္ဆာ 1 အဖြေ

```php
<?php
$name  = "田中太郎";
$age   = 20;
$hobby = "プログラミング";

echo "名前：{$name}<br>";
echo "年齢：{$age}歳<br>";
echo "趣味：{$hobby}<br>";
?>
```

---

## ပုစ္ဆာ 2 အဖြေ

```php
<?php
$fruits = ["りんご", "みかん", "ぶどう", "バナナ"];

echo "<ul>";
foreach ($fruits as $fruit) {
    echo "<li>{$fruit}</li>";
}
echo "</ul>";
?>
```

---

## ပုစ္ဆာ 3 အဖြေ

```php
<?php
$total = 0;

for ($i = 1; $i <= 10; $i++) {
    $total += $i;
}

echo "合計: " . $total;  // → 合計: 55 (စုစုပေါင်း: 55)
?>
```

## ပုစ္ဆာ 4 အဖြေ（အဆင့်မြင့်）

```php
<?php
for ($dan = 2; $dan <= 5; $dan++) {
    for ($i = 1; $i <= 9; $i++) {
        echo "{$dan}×{$i}=" . ($dan * $i) . "  ";
    }
    echo "<br>";
}
?>
```
